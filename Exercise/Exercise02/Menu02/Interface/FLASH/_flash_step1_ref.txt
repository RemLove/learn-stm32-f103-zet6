【0】这题的完整路线（别跳步）
  第 1 步（本次）：把 SPI2 跑通，读到 0xEF4017
  第 2 步：擦除 / 写入 / 读取 + BUSY 轮询
  第 3 步：黑匣子记录框架（定长记录 + 断电续写）+ 加进菜单

  "黑匣子"是什么：板子掉电后数据还在，上电能把历史记录读出来。
  它和 EEPROM 不一样 —— SPI Flash 不能"改一个字节"，写之前必须先把整段擦成 0xFF。
  所以第三步的重点不是"写"，而是"怎么在只能整段擦的介质上做追加记录"。


【1】硬件映射（板子上焊死的，别改）
  信号     引脚      MCU 侧配置
  -------- --------  ------------------------------
  CS       PB12      普通推挽输出，空闲为高（软件片选）
  SCK      PB13      SPI2_SCK，复用推挽
  MISO     PB14      浮空输入
  MOSI     PB15      SPI2_MOSI，复用推挽

  SPI2 挂在 APB1 上，Menu02 里 APB1 = 36MHz。
  所以 SPI 波特率 = 36MHz / 分频系数，没有别的来源。

  为什么要单独配 CS：SPI 硬件只管时钟和数据，片选是"选哪个从机"，
  总线上可能挂多个器件，所以 CS 一律当普通 GPIO 自己拉（NSS = SOFT）。

  JEDEC ID 对照（0x9F 读回来 3 个字节：厂商 + 容量高 + 容量低）
  0xEF4014 -> W25Q80    1MB
  0xEF4015 -> W25Q16    2MB
  0xEF4016 -> W25Q32    4MB
  0xEF4017 -> W25Q64    8MB    <-- 霸道板载这颗
  0xEF4018 -> W25Q128   16MB
  厂商字节 0xEF = Winbond（华邦）

【4.2】Core/Src/spi.c

#include "spi.h"

SPI_HandleTypeDef hspi2;

void MX_SPI2_Init(void)
{
  hspi2.Instance = SPI2;
  hspi2.Init.Mode = SPI_MODE_MASTER;              /* 主机 */
  hspi2.Init.Direction = SPI_DIRECTION_2LINES;    /* 全双工 */
  hspi2.Init.DataSize = SPI_DATASIZE_8BIT;        /* 8 位一帧 */
  hspi2.Init.CLKPolarity = SPI_POLARITY_LOW;      /* CPOL = 0，空闲低 */
  hspi2.Init.CLKPhase = SPI_PHASE_1EDGE;          /* CPHA = 0，第一个边沿采样 -> 模式 0 */
  hspi2.Init.NSS = SPI_NSS_SOFT;                  /* 片选自己拉 */
  hspi2.Init.BaudRatePrescaler = SPI_BAUDRATEPRESCALER_8;  /* 36MHz/8 = 4.5MHz，先慢后快 */
  hspi2.Init.FirstBit = SPI_FIRSTBIT_MSB;         /* 高位先出 */
  hspi2.Init.TIMode = SPI_TIMODE_DISABLE;
  hspi2.Init.CRCCalculation = SPI_CRCCALCULATION_DISABLE;
  hspi2.Init.CRCPolynomial = 10;

  if (HAL_SPI_Init(&hspi2) != HAL_OK)
  {
    Error_Handler();
  }
}

/* HAL_SPI_Init() 内部会回调这个弱函数，引脚和时钟都在这儿配。
   注意：Menu02 的 stm32f1xx_hal_msp.c 里没有 HAL_SPI_MspInit，
   所以放 spi.c 里，不会重复定义。 */
void HAL_SPI_MspInit(SPI_HandleTypeDef *spiHandle)
{
  GPIO_InitTypeDef GPIO_InitStruct = {0};

  if (spiHandle->Instance == SPI2)
  {
    __HAL_RCC_SPI2_CLK_ENABLE();      /* SPI2 时钟不在这句就全白干 */
    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* PB13 = SCK，PB15 = MOSI：复用推挽（时钟和数据都由片内外设推） */
    GPIO_InitStruct.Pin = GPIO_PIN_13 | GPIO_PIN_15;
    GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_HIGH;
    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);

    /* PB14 = MISO：F1 上"复用输入"就是浮空输入，别写成 AF_PP，
       写错会两个器件同时往这根线上推，读回来全是垃圾 */
    GPIO_InitStruct.Pin = GPIO_PIN_14;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);

    /* PB12 = CS：普通推挽输出。先把电平写成高，再切成输出模式，
       否则配置那一瞬间引脚是低，等于给芯片发了个假片选 */
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_SET);
    GPIO_InitStruct.Pin = GPIO_PIN_12;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_HIGH;
    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);
  }
}

#ifndef __SPI_FLASH_H
#define __SPI_FLASH_H

#include "main.h"

/* W25Q 系列指令（手册 Instruction Set 那一章，先只记这三个） */
#define W25Q_CMD_JEDEC_ID       0x9F   /* 读厂商+容量 ID，3 字节 */
#define W25Q_CMD_READ_SR1       0x05   /* 读状态寄存器 1 */
#define W25Q_CMD_WRITE_ENABLE   0x06   /* 写使能，擦/写之前必须发 */

void     SPI_Flash_Init(void);
uint32_t SPI_Flash_ReadID(void);
uint8_t  SPI_Flash_ReadSR1(void);
uint8_t  SPI_Flash_IsBusy(void);

#endif

【4.4】Interface/FLASH/spi_flash.c
------------------------------------------------------------------
#include "spi_flash.h"
#include "spi.h"

/* 片选：写死 PB12。以后要换引脚只改这两行 */
#define FLASH_CS_LOW()   HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_RESET)
#define FLASH_CS_HIGH()  HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_SET)

/* 底层：全双工交换 1 个字节。
   SPI 没有"只读"这回事 —— 想收 1 个字节，必须同时发 1 个字节出去，
   发什么无所谓（习惯发 0xFF），时钟由主机发数据产生。 */
static uint8_t SPI_TxRx(uint8_t tx)
{
  uint8_t rx = 0xFF;

  if (HAL_SPI_TransmitReceive(&hspi2, &tx, &rx, 1, 100) != HAL_OK)
  {
    return 0xFF;          /* 超时就返回 0xFF，让上层读到明显的错值 */
  }
  return rx;
}

/* 初始化：把片选拉到空闲高电平，等芯片上电稳定 */
void SPI_Flash_Init(void)
{
  FLASH_CS_HIGH();
  HAL_Delay(10);
}

/* 读 JEDEC ID：拉低 CS -> 0x9F -> 收 3 字节 -> 拉高 CS */
uint32_t SPI_Flash_ReadID(void)
{
  uint32_t id = 0;

  FLASH_CS_LOW();

  SPI_TxRx(W25Q_CMD_JEDEC_ID);
  id  = (uint32_t)SPI_TxRx(0xFF) << 16;   /* 厂商 ID，应为 0xEF */
  id |= (uint32_t)SPI_TxRx(0xFF) << 8;    /* 容量高字节，0x40 */
  id |= (uint32_t)SPI_TxRx(0xFF);         /* 容量低字节，0x17 = 64Mbit */

  FLASH_CS_HIGH();
  return id;
}

/* 读状态寄存器 1：bit0 = BUSY，bit1 = WEL（写使能锁存） */
uint8_t SPI_Flash_ReadSR1(void)
{
  uint8_t sr;

  FLASH_CS_LOW();
  SPI_TxRx(W25Q_CMD_READ_SR1);
  sr = SPI_TxRx(0xFF);
  FLASH_CS_HIGH();

  return sr;
}

/* BUSY 位为 1 说明芯片正在擦/写，这时候发任何指令它都不理你 */
uint8_t SPI_Flash_IsBusy(void)
{
  return (uint8_t)(SPI_Flash_ReadSR1() & 0x01);
}

【5】怎么算通过
 串口（115200，VOFA+ 或串口助手）应该看到：

   S1: before LCD_Init
   S2: LCD_Init done, ID = 0x9341
   FLASH JEDEC = 0xEF4017
   FLASH SR1   = 0x00
   S5: menu ready

 0xEF4017 = W25Q64 8MB，和板子对得上就说明 SPI2 通了。
 SR1 = 0x00 说明芯片空闲（bit0 BUSY=0，bit1 WEL=0）。


【6】读到怪值时先查哪儿（照这个顺序，别乱试）

  读到的值        说明什么                  先查
  ------------    ------------------------  ----------------------------------
  0x000000        MISO 一直是低              CS 有没有真的拉低 / MISO 是不是短地
                                             / 芯片是不是没焊上
  0xFFFFFF        MISO 一直是高              芯片没供电 / CS 没接 / SPI2 时钟没开
                                             / MOSI 没发出去
  0x0000FF        只收到一个字节              片选提前拉高了，或者循环次数写错
  0xEF40xx        厂商对、容量不对            查芯片型号，可能不是 W25Q64
  0xEF4017 但偶发错 速率或时序临界            分频改 16（2.25MHz）再试
  一直 0xFF 且 LCD 也花屏                    电源/地没共好，先量 3.3V

  另一个快速判断法：把 SCK 用示波器/逻辑分析仪看一眼，
  发指令时应该有 8 个脉冲；一个脉冲都没有 = 时钟没开或 SPI 没使能。


【7】这一步最容易踩的坑（我按踩到概率排的）

 1. 片选没人管。NSS 配成 SOFT 之后，HAL 不会帮你拉 CS，必须自己来。
    忘了拉低 -> 芯片根本不睬你，读回 0xFF。
 2. 用 HAL_SPI_Transmit 去读 ID。只发不收，返回的 rx 是 0。
    读数据一律 HAL_SPI_TransmitReceive（或者 Transmit 完再 Receive，
    但那样 CS 中间会断，多字节读会错位）。
 3. 每读一个字节都要先发一个 0xFF。SPI 的时钟是"发"出来的，
    不发就没有时钟，芯片也不会吐数据。
 4. PB14 写成 GPIO_MODE_AF_PP。MISO 是输入，写成输出会打架。
 5. __HAL_RCC_SPI2_CLK_ENABLE() 漏了。表现是 HAL_SPI_Init 返回 OK，
    但 SCK 上一个脉冲都没有。
 6. 忘了打开 HAL_SPI_MODULE_ENABLED。编译报 SPI_HandleTypeDef 未定义。
 7. Keil 里忘了加 Include Paths 里的 ..\Interface\FLASH。
 8. printf 打中文。Keil 源文件是 GBK，串口助手按 UTF-8 显示就是乱码
    （你旧工程 ex08 里那个 "оƬID" 就是这坑）。调试输出全用英文。
 9. 一开始就想跑 18MHz（分频 2）。杜邦线/排针一长，18MHz 就开始出错，
    排错时先把速度降到 4.5MHz，通了再往上提。
 10. 状态寄存器读一次就下结论。擦除要几十毫秒，BUSY 期间读出来是 0xFF，
     第二步写擦除的时候必须 while 轮询 BUSY，不能只读一次。

【9】下一步（第二步）会用到的东西，先有个印象
  - 0x06 WriteEnable：每次擦/写之前单独发一次，CS 拉高才算生效
  - 0x05 读 SR1 轮询 BUSY
  - 0x20 扇区擦除（4KB），地址 24 位，发完 CS 拉高才开始擦
  - 0x02 页编程（256 字节一页，跨页会回卷到页首，这是最容易错的地方）
  - 0x03 读数据（可以连续读，地址自动加）
  这些等第一步通了再讲，先把 ID 读对。
