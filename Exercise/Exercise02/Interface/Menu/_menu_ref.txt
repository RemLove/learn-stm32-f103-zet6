【硬件映射】注意这个，容易混
  菜单里的 "LED1"  ->  野火板 LED0  ->  PB5  ->  宏 LED0_Pin / LED0_GPIO_Port
  菜单里的 "LED2"  ->  野火板 LED1  ->  PE5  ->  宏 LED1_Pin / LED1_GPIO_Port
  CubeMX 里的 Label 填的是 LED0/LED1，所以代码里必须用 LED0_xxx / LED1_xxx。

【极性】野火板 LED 低电平点亮
  GPIO_PIN_RESET = 亮
  GPIO_PIN_SET   = 灭

【按键在每层的含义】
  KEY_UP / KEY1    ：上下移动光标
  KEY0 短按        ：主菜单->进LED列表；LED列表->进on/off；on/off->执行亮灭
  KEY0 长按        ：逐层返回；主菜单长按无操作

 menu.h

#ifndef __MENU_H
#define __MENU_H

#include "lcd.h"
#include "LCD_Front.h"
#include "key.h"
#include "usart.h"

/* 菜单层级 */
typedef enum
{
    LVL_MAIN = 0,      /* 主菜单：1.LED 2.PWM 3.TIME 4.INFO */
    LVL_LED,           /* LED 列表：LED1 / LED2 */
    LVL_LED_OP         /* 亮灭选择：on / off */
} menu_lvl_t;

void Menu_Init(void);      /* 初始化，进主菜单 */
void Menu_Scan(void);      /* 主循环里调，处理按键 */
void Menu_Draw(void);      /* 重画主菜单 */

#endif


================================================================
 menu.c
================================================================

#include "menu.h"

#define MENU_COUNT      4
#define LED_COUNT       2
#define OP_COUNT        2

/* 前置声明：Scna_Main 里要用 Draw_LedList，它在后面才定义 */
static void Draw_LedList(void);
static void Draw_LedOp(void);

/* ===== 主菜单文字 ===== */
static uint8_t *menu_text[MENU_COUNT] = {
    (uint8_t *)"1.LED",
    (uint8_t *)"2.PWM",
    (uint8_t *)"3.TIME",
    (uint8_t *)"4.INFO"
};

/* ===== 各层 y 坐标表，类型必须统一 uint16_t ===== */
static const uint16_t menu_y[MENU_COUNT] = {100, 130, 160, 190};
static const uint16_t led_y[LED_COUNT]   = {120, 160};
static const uint16_t op_y[OP_COUNT]     = {120, 160};

/* ===== 状态 ===== */
static menu_lvl_t level  = LVL_MAIN;
static uint8_t    sel[3] = {0, 0, 0};   /* 每层选中的项 */

/* ===== 光标：在 (0, y) 画或擦 "->" ===== */
static void Cursor_Set(uint16_t y, uint16_t color)
{
    LCD_WriteAsiciStr(0, y, 16, color, BLACK, (uint8_t *)"->");
}

/* ===== 通用上下移动 =====
   y_table : 这一层的坐标表
   index   : 指向这一层的 sel[n]
   count   : 这一层共几项
   evt     : KEY_UP 或 KEY_DOWN
*/
static void Cursor_Move(const uint16_t *y_table, uint8_t *index,
                        uint8_t count, Key_evt_t evt)
{
    if(count == 0) return;                      /* 防除零 */

    Cursor_Set(y_table[*index], BLACK);         /* 1.擦旧 */

    if(evt == KEY_DOWN)                         /* 2.算新 */
    {
        *index = (uint8_t)((*index + 1) % count);
    }
    else if(evt == KEY_UP)
    {
        *index = (*index == 0) ? (uint8_t)(count - 1) : (uint8_t)(*index - 1);
    }

    Cursor_Set(y_table[*index], RED);           /* 3.画新，RED！不是 BLACK */
}

/* ===== LED 控制 =====
   which: 0=菜单里的LED1(PB5)  1=菜单里的LED2(PE5)
   on   : 1=点亮 0=熄灭
*/
static void LED_Set(uint8_t which, uint8_t on)
{
    GPIO_TypeDef *port = (which == 0) ? LED0_GPIO_Port : LED1_GPIO_Port;
    uint16_t      pin  = (which == 0) ? LED0_Pin       : LED1_Pin;

    /* 低电平点亮 */
    HAL_GPIO_WritePin(port, pin, on ? GPIO_PIN_RESET : GPIO_PIN_SET);
}

/* ============================================================
   第一层：主菜单
   ============================================================ */
void Menu_Draw(void)
{
    LCD_BGOn();
    LCD_ClearAll(BLACK);

    LCD_WriteAsiciStr(0, 30, 24, WHITE, BLACK, (uint8_t *)"Menu");
    LCD_DrawLine(30, 80, 210, 80, 1, GRAY);

    for(uint8_t i = 0; i < MENU_COUNT; i++)
    {
        LCD_WriteAsiciStr(20, menu_y[i], 16, WHITE, BLACK, menu_text[i]);
    }

    LCD_DrawLine(30, 226, 210, 226, 1, GRAY);

    Cursor_Set(menu_y[sel[0]], RED);            /* 初始光标 */
}

static void Scan_Main(void)
{
    if(key_evt == KEY_UP || key_evt == KEY_DOWN)
    {
        Cursor_Move(menu_y, &sel[0], MENU_COUNT, key_evt);
        key_evt = KEY_NONE;
        return;
    }

    switch(key_evt)
    {
        case KEY_SHORT:
            key_evt = KEY_NONE;
            if(sel[0] == 0)                     /* 选了 1.LED */
            {
                level  = LVL_LED;
                sel[1] = 0;
                Draw_LedList();
            }
            /* 2.PWM / 3.TIME / 4.INFO 以后在这里加 */
            break;

        case KEY_LONG:
            key_evt = KEY_NONE;                 /* 主菜单长按无操作 */
            break;

        default:
            key_evt = KEY_NONE;
            break;
    }
}

/* ============================================================
   第二层：LED 列表
   ============================================================ */
static void Draw_LedList(void)
{
    LCD_ClearAll(BLACK);

    LCD_WriteAsiciStr(0, 30, 24, YELLOW, BLACK, (uint8_t *)"LED");
    LCD_DrawLine(30, 80, 210, 80, 1, GRAY);

    LCD_WriteAsiciStr(20, led_y[0], 16, WHITE, BLACK, (uint8_t *)"1. LED1");
    LCD_WriteAsiciStr(20, led_y[1], 16, WHITE, BLACK, (uint8_t *)"2. LED2");

    LCD_DrawLine(30, 210, 210, 210, 1, GRAY);
    LCD_WriteAsiciStr(0, 290, 16, GRAY, BLACK, (uint8_t *)"long OK = back");

    Cursor_Set(led_y[sel[1]], RED);
}

static void Scan_LedList(void)
{
    if(key_evt == KEY_UP || key_evt == KEY_DOWN)
    {
        Cursor_Move(led_y, &sel[1], LED_COUNT, key_evt);
        key_evt = KEY_NONE;
        return;
    }

    switch(key_evt)
    {
        case KEY_SHORT:
            key_evt = KEY_NONE;
            level  = LVL_LED_OP;
            sel[2] = 0;                         /* 默认停在 on */
            Draw_LedOp();
            break;

        case KEY_LONG:
            key_evt = KEY_NONE;
            level = LVL_MAIN;
            Menu_Draw();                        /* sel[0] 保留，光标回原位 */
            break;

        default:
            key_evt = KEY_NONE;
            break;
    }
}

/* ============================================================
   第三层：on / off
   ============================================================ */
static void Draw_LedOp(void)
{
    LCD_ClearAll(BLACK);

    /* 标题显示当前操作哪个灯 */
    LCD_WriteAsiciStr(0, 30, 24, YELLOW, BLACK,
                      (uint8_t *)((sel[1] == 0) ? "LED1" : "LED2"));
    LCD_DrawLine(30, 80, 210, 80, 1, GRAY);

    LCD_WriteAsiciStr(20, op_y[0], 16, WHITE, BLACK, (uint8_t *)"1. on");
    LCD_WriteAsiciStr(20, op_y[1], 16, WHITE, BLACK, (uint8_t *)"2. off");

    LCD_DrawLine(30, 210, 210, 210, 1, GRAY);
    LCD_WriteAsiciStr(0, 290, 16, GRAY, BLACK, (uint8_t *)"long OK = back");

    Cursor_Set(op_y[sel[2]], RED);
}

static void Scan_LedOp(void)
{
    if(key_evt == KEY_UP || key_evt == KEY_DOWN)
    {
        Cursor_Move(op_y, &sel[2], OP_COUNT, key_evt);
        key_evt = KEY_NONE;
        return;
    }

    switch(key_evt)
    {
        case KEY_SHORT:                         /* 执行亮/灭 */
            key_evt = KEY_NONE;
            LED_Set(sel[1], (sel[2] == 0));     /* sel[2]==0 是 on */
            break;

        case KEY_LONG:
            key_evt = KEY_NONE;
            level = LVL_LED;
            Draw_LedList();
            break;

        default:
            key_evt = KEY_NONE;
            break;
    }
}

/* ============================================================
   入口
   ============================================================ */
void Menu_Init(void)
{
    level  = LVL_MAIN;
    sel[0] = 0;
    sel[1] = 0;
    sel[2] = 0;
    Menu_Draw();
}

void Menu_Scan(void)
{
    switch(level)
    {
        case LVL_MAIN:   Scan_Main();    break;
        case LVL_LED:    Scan_LedList(); break;
        case LVL_LED_OP: Scan_LedOp();   break;
        default:         level = LVL_MAIN; break;
    }
}


================================================================
 和 main.c 的配合
================================================================

main.c 里：

  #include "menu.h"

  /* USER CODE BEGIN 2 */
  LCD_Init();
  ...
  Menu_Init();          // 画主菜单

  /* while (1) 里 */
  Key_Scan();           // 扫按键，产生 key_evt
  Menu_Scan();          // 消费 key_evt，更新菜单

注意：Menu_Scan 必须在 Key_Scan 之后调，顺序反了菜单不响应。


================================================================
 测试路径
================================================================

上电        -> 主菜单，"->" 在 1.LED
KEY0 短按   -> 进 LED 列表（LED1/LED2）
KEY1        -> 光标移到 LED2
KEY0 短按   -> 进 on/off，标题显示 LED2
KEY1        -> 光标到 off
KEY0 短按   -> PE5 灭
KEY_UP      -> 光标回 on
KEY0 短按   -> PE5 亮
KEY0 长按   -> 回 LED 列表，光标还在 LED2      <-- 重点验证
KEY0 长按   -> 回主菜单，光标还在 1.LED        <-- 重点验证
